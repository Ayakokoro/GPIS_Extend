cmake_minimum_required(VERSION 3.20)
project(gpis_nvdb_to_vdb LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(OpenVDB REQUIRED)
find_package(nanovdb CONFIG REQUIRED)
find_package(nlohmann_json CONFIG REQUIRED)
find_package(TBB CONFIG REQUIRED)
add_executable(gpis_nvdb_to_vdb convert.cpp)
target_compile_definitions(gpis_nvdb_to_vdb PRIVATE NANOVDB_USE_OPENVDB NOMINMAX)
target_link_libraries(gpis_nvdb_to_vdb PRIVATE
    OpenVDB::openvdb nanovdb::nanovdb nlohmann_json::nlohmann_json TBB::tbb)
if(MSVC)
    target_compile_options(gpis_nvdb_to_vdb PRIVATE /EHsc /utf-8 /bigobj)
endif()
